## C++  (`src/run_file/`)

**main.cpp**

#include<iostream>

int main(){
	std::cout << "Hello World!" << std::endl;

**Makefile**

```makefile
CC = g++
BINNAME = main.exe

$(BINNAME): main.cpp
	$(CC) -o $@ $<

clean:
	rm -f *.exe *.txt *.sh
```

**run_prog.sh**

```bash
#!/usr/bin/env bash
make_run_file() {
cat > $1.sh <<%EOF%
#!/bin/bash
#SBATCH --job-name=$1
#SBATCH --output="out.txt"
#SBATCH --error="err.txt"
#SBATCH --time=00:00:30
#SBATCH --mem=1M

make
./main.exe
%EOF%
}

name=run_file

make_run_file ${name}
sbatch ${name}.sh
```
